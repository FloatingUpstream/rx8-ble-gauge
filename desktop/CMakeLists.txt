cmake_minimum_required(VERSION 3.20)
project(rx8gauge_desktop LANGUAGES C CXX)
set(CMAKE_C_STANDARD 17)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(LVGL_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../.local/aux/upstream/lvgl"
    CACHE PATH "Pinned LVGL 9.5.0 source")
set(LV_BUILD_CONF_PATH "${CMAKE_CURRENT_SOURCE_DIR}/lv_conf.h" CACHE PATH "" FORCE)
set(CONFIG_LV_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
find_package(PkgConfig REQUIRED)
pkg_check_modules(SDL2 REQUIRED IMPORTED_TARGET sdl2)
find_package(PNG REQUIRED)
add_subdirectory("${LVGL_SOURCE_DIR}" "${CMAKE_BINARY_DIR}/lvgl")
target_link_libraries(lvgl PUBLIC PkgConfig::SDL2)
add_library(gauge_ui STATIC ../ui/gauge_ui.c ../ui/adapter_ui.c)
target_include_directories(gauge_ui PUBLIC ../ui ../protocol)
target_link_libraries(gauge_ui PUBLIC lvgl::lvgl)
add_library(elm_decode STATIC ../protocol/elm_decode.c ../protocol/elm_session.c)
target_include_directories(elm_decode PUBLIC ../protocol)
add_executable(gauge_desktop main.c elm_tcp.c)
target_link_libraries(gauge_desktop PRIVATE gauge_ui elm_decode PNG::PNG m)
foreach(target gauge_ui elm_decode gauge_desktop)
    target_compile_options(${target} PRIVATE -Wall -Wextra -Werror)
endforeach()
enable_testing()
add_executable(elm_decode_test ../tests/elm_decode_test.c)
target_link_libraries(elm_decode_test PRIVATE elm_decode)
add_test(NAME elm_decode COMMAND elm_decode_test)
add_executable(elm_session_test ../tests/elm_session_test.c)
target_link_libraries(elm_session_test PRIVATE elm_decode)
add_test(NAME elm_session COMMAND elm_session_test)
add_executable(adapter_identity_test ../tests/adapter_identity_test.c)
target_link_libraries(adapter_identity_test PRIVATE elm_decode)
add_test(NAME adapter_identity COMMAND adapter_identity_test)
foreach(test elm_fault elm_soak)
    add_executable(${test}_test ../tests/${test}_test.c)
    target_link_libraries(${test}_test PRIVATE elm_decode)
    add_test(NAME ${test} COMMAND ${test}_test)
endforeach()
add_test(NAME render_smoke COMMAND gauge_desktop --scenario normal
         --snapshot "${CMAKE_BINARY_DIR}/smoke.png")
set_tests_properties(render_smoke PROPERTIES ENVIRONMENT "SDL_VIDEODRIVER=dummy")
foreach(target elm_decode_test elm_session_test adapter_identity_test elm_fault_test elm_soak_test)
    target_compile_options(${target} PRIVATE -Wall -Wextra -Werror)
endforeach()
option(GAUGE_SANITIZERS "Instrument project-owned host code with ASan/UBSan" OFF)
if(GAUGE_SANITIZERS)
    foreach(target gauge_ui elm_decode gauge_desktop elm_decode_test elm_session_test
                   adapter_identity_test elm_fault_test elm_soak_test)
        target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
        target_link_options(${target} PRIVATE -fsanitize=address,undefined)
    endforeach()
endif()
